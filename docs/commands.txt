#####
BUID ON LOCAL
#####
docker compose build

#####
RUN ON LOCAL
#####
docker compose up 

#####
BUID ON SERVER
#####
docker compose -f compose.prod.yaml build

#####
RUN ON SERVER
#####
docker compose -f compose.prod.yaml up -d

#####
CHECKING DOCKER LOGS
#####
docker logs -f {container name}

######
FOR GIVING PREVILATE TO MYSQL USER TO A DB
######
docker exec -it avnik-mysql-prod mysql -uroot -p -e "
ALTER USER 'avnik_user'@'%' IDENTIFIED BY 'avnik_password';
GRANT ALL PRIVILEGES ON avnikdb.* TO 'avnik_user'@'%';
FLUSH PRIVILEGES;"
